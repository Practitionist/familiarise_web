 Phase 2: Prisma Schema Enhancements (1 hour)

 Add new models and fields:

 // New models
 model Refund {
   id              String        @id @default(uuid())
   amount          Int
   currency        String
   reason          String?
   status          RefundStatus
   refundId        String        @unique  // Gateway refund ID
   paymentGateway  PaymentGateway
   payment         Payment       @relation(...)
   paymentId       String
   createdAt       DateTime      @default(now())
   updatedAt       DateTime      @updatedAt
 }

 enum RefundStatus {
   PENDING
   SUCCEEDED
   FAILED
   CANCELLED
 }

 model Dispute {
   id              String        @id @default(uuid())
   amount          Int
   currency        String
   reason          String
   status          DisputeStatus
   disputeId       String        @unique  // Gateway dispute ID
   evidence        Json?         // Evidence submitted
   paymentGateway  PaymentGateway
   payment         Payment       @relation(...)
   paymentId       String
   createdAt       DateTime      @default(now())
   updatedAt       DateTime      @updatedAt
 }

 enum DisputeStatus {
   WARNING_NEEDS_RESPONSE
   WARNING_UNDER_REVIEW
   WARNING_CLOSED
   NEEDS_RESPONSE
   UNDER_REVIEW
   CHARGE_REFUNDED
   WON
   LOST
 }

 // Update Payment model
 model Payment {
   // ... existing fields
   isMockPayment   Boolean       @default(false)  // NEW
   refunds         Refund[]      // NEW
   disputes        Dispute[]     // NEW
 }
