### **Step 1: Connect to Database & Find Test Consultant**

**Supabase MCP Query:**
```sql
-- Find a consultant with availability who has pending requests
SELECT 
  cp.id as consultant_profile_id,
  u.id as user_id,
  u.name,
  u.email,
  cp.scheduleType,
  COUNT(DISTINCT sw.id) as weekly_slots_count,
  COUNT(DISTINCT sc.id) as custom_slots_count,
  COUNT(DISTINCT c.id) as pending_consultations,
  COUNT(DISTINCT s.id) as pending_subscriptions
FROM "ConsultantProfile" cp
JOIN users u ON cp."userId" = u.id
LEFT JOIN "SlotOfAvailabilityWeekly" sw ON sw."consultantProfileId" = cp.id
LEFT JOIN "SlotOfAvailabilityCustom" sc ON sc."consultantProfileId" = cp.id
LEFT JOIN "ConsultationPlan" cplan ON cplan."consultantProfileId" = cp.id
LEFT JOIN "Consultation" c ON c."consultationPlanId" = cplan.id AND c.requestStatus = 'PENDING'
LEFT JOIN "SubscriptionPlan" splan ON splan."consultantProfileId" = cp.id
LEFT JOIN "Subscription" s ON s."subscriptionPlanId" = splan.id AND s.requestStatus = 'PENDING'
GROUP BY cp.id, u.id, u.name, u.email, cp.scheduleType
HAVING COUNT(DISTINCT sw.id) > 0 OR COUNT(DISTINCT sc.id) > 0
LIMIT 5;
```

### **Step 3: Check Existing Appointments**

**Supabase MCP Query:**
```sql
-- Get all booked slots for this consultant
SELECT 
  a.id as appointment_id,
  a.appointmentType,
  sa.id as slot_id,
  sa.slotStartTimeInUTC,
  sa.slotEndTimeInUTC,
  sa.isTentative,
  TO_CHAR(sa.slotStartTimeInUTC, 'YYYY-MM-DD') as booking_date,
  TO_CHAR(sa.slotStartTimeInUTC, 'HH24:MI') as booking_time,
  -- Get participant info
  consee.name as consultee_name,
  COALESCE(
    cons.requestStatus::text,
    subs.requestStatus::text,
    cls.status::text,
    web.status::text
  ) as status
FROM "Appointment" a
JOIN "SlotOfAppointment" sa ON sa."appointmentId" = a.id
JOIN "_SlotOfAppointmentToUser" su ON su."A" = sa.id
JOIN users u ON su."B" = u.id
LEFT JOIN "Consultation" cons ON a."consultationId" = cons.id
LEFT JOIN "ConsulteeProfile" cp_consee ON cons."requestedById" = cp_consee.id
LEFT JOIN users consee ON cp_consee."userId" = consee.id
LEFT JOIN "Subscription" subs ON a."subscriptionId" = subs.id
LEFT JOIN "Class" cls ON a."classId" = cls.id
LEFT JOIN "Webinar" web ON a."webinarId" = web.id
WHERE u."consultantProfileId" = '[INSERT_CONSULTANT_PROFILE_ID]'
  AND sa.slotStartTimeInUTC > NOW()
ORDER BY sa.slotStartTimeInUTC
LIMIT 20;
```

### **Step 11: Check for Consecutive Slot Violation**

**Supabase MCP Query:**
```sql
-- Verify slots are consecutive (30 minutes apart)
WITH slot_intervals AS (
  SELECT 
    sa.slotStartTimeInUTC,
    sa.slotEndTimeInUTC,
    LAG(sa.slotEndTimeInUTC) OVER (ORDER BY sa.slotStartTimeInUTC) as previous_end_time,
    EXTRACT(EPOCH FROM (sa.slotStartTimeInUTC - LAG(sa.slotEndTimeInUTC) OVER (ORDER BY sa.slotStartTimeInUTC))) as gap_seconds
  FROM "SlotOfAppointment" sa
  JOIN "Appointment" a ON sa."appointmentId" = a.id
  WHERE a."consultationId" = '[CONSULTATION_ID]'
  ORDER BY sa.slotStartTimeInUTC
)
SELECT 
  *,
  CASE 
    WHEN gap_seconds IS NULL THEN 'FIRST_SLOT'
    WHEN gap_seconds <= 1 THEN 'CONSECUTIVE'
    ELSE 'GAP_DETECTED'
  END as slot_status
FROM slot_intervals;
```

**Expected Output:**
```
| slotStartTimeInUTC       | slotEndTimeInUTC         | gap_seconds | slot_status   |
|--------------------------|--------------------------|-------------|---------------|

| 2025-10-13 10:30:00+00   | 2025-10-13 11:00:00+00   | 0           | CONSECUTIVE   |
| 2025-10-13 11:00:00+00   | 2025-10-13 11:30:00+00   | 0           | CONSECUTIVE   |
```

### **Step 14: Verify Weekly Limit Enforcement**

**Supabase MCP Query:**
```sql
-- Check slots per week
WITH weekly_distribution AS (
  SELECT 
    DATE_TRUNC('week', sa.slotStartTimeInUTC) as week_start,
    DATE(sa.slotStartTimeInUTC) as booking_date,
    COUNT(DISTINCT DATE(sa.slotStartTimeInUTC)) as unique_days_in_week,
    COUNT(*) as total_slots,
    COUNT(*) / 2 as total_calls, -- 2 slots = 1 call (assuming 1-hour sessions)
    array_agg(DISTINCT TO_CHAR(sa.slotStartTimeInUTC, 'YYYY-MM-DD HH24:MI')) as all_bookings
  FROM "SlotOfAppointment" sa
  JOIN "Appointment" a ON sa."appointmentId" = a.id
  WHERE a."subscriptionId" = '[SUBSCRIPTION_ID]'
  GROUP BY week_start
  ORDER BY week_start
)
SELECT 
  week_start,
  total_calls,
  unique_days_in_week,
  all_bookings,
  CASE 
    WHEN total_calls > 2 THEN '⚠️ EXCEEDS LIMIT'
    WHEN unique_days_in_week < total_calls THEN '⚠️ MULTIPLE CALLS SAME DAY'
    ELSE '✅ VALID'
  END as validation_status
FROM weekly_distribution;
```

**Expected Output:**
```
| week_start | total_calls | unique_days | validation_status |
|------------|-------------|-------------|-------------------|
| 2025-10-13 | 2           | 2           | ✅ VALID          |
| 2025-10-20 | 2           | 2           | ✅ VALID          |
| 2025-10-27 | 2           | 2           | ✅ VALID          |
| 2025-11-03 | 2           | 2           | ✅ VALID          |
```

**If you see:**
- `total_calls > 2`: ⚠️ **BUG** - Weekly limit violated!
- `unique_days < total_calls`: ⚠️ **BUG** - Multiple calls on same day!
