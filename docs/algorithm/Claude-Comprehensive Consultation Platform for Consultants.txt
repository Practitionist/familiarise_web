interface CheckoutSession {

// app/api/checkout/register/[appointmentType]/route.ts
export async function POST(
  req: Request,
  { params }: { params: { appointmentType: AppointmentsType } }
) {
  const { planId, slotIds, paymentGateway = 'STRIPE' } = await req.json();

  try {
    return await prisma.$transaction(async (tx) => {
      // 1. Check slot availability with lock
      const existingSlot = await tx.slotOfAppointment.findFirst({
        where: {
          slotStartTimeInUTC,
          slotEndTimeInUTC,
          OR: [
            { isTentative: false },
            {
              isTentative: true,
              expiresAt: { gt: new Date() }
            }
          ]
        }
      });

      if (existingSlot) {
        throw new Error('SLOT_UNAVAILABLE');
      }

      // 2. Create temporary slot with lock
      const slot = await tx.slotOfAppointment.create({
        data: {
          slotStartTimeInUTC,
          slotEndTimeInUTC,
          isTentative: true,
          expiresAt: new Date(Date.now() + 15 * 60 * 1000)
        }
      });

      // 3. Create checkout session
      const checkoutSession = await tx.checkoutSession.create({
        data: {
          slotId: slot.id,
          appointmentType: params.appointmentType,
          planId,
          status: 'PENDING',
          expiresAt: new Date(Date.now() + 15 * 60 * 1000),
          retryCount: 0
        }
      });

      // 4. Create payment intent
      const paymentIntent = await createPaymentIntent({
        amount,
        currency: 'USD',
        metadata: {
          checkoutSessionId: checkoutSession.id,
          slotId: slot.id
        }
      });

      // 5. Update session with payment info
      await tx.checkoutSession.update({
        where: { id: checkoutSession.id },
        data: { paymentIntentId: paymentIntent.id }
      });

      return {
        clientSecret: paymentIntent.client_secret,
        sessionId: checkoutSession.id
      };
    });
  } catch (error) {
    if (error.message === 'SLOT_UNAVAILABLE') {
      return new Response('Slot no longer available', { status: 409 });
    }
    throw error;
  }
}

// Webhook handler with retry logic
export async function POST(req: Request) {
  const event = await validateWebhookEvent(req);
  
  if (event.type === 'payment_intent.succeeded') {
    const { checkoutSessionId } = event.metadata;
    
    try {
      await prisma.$transaction(async (tx) => {
        // 1. Get checkout session
        const session = await tx.checkoutSession.findUnique({
          where: { id: checkoutSessionId }
        });

        if (!session) {
          throw new Error('INVALID_SESSION');
        }

        // 2. Check slot availability again
        const slot = await tx.slotOfAppointment.findUnique({
          where: { id: session.slotId }
        });

        if (!slot || !slot.isTentative) {
          // Slot was taken - initiate refund
          await initiateRefund(session.paymentIntentId);
          throw new Error('SLOT_TAKEN');
        }

        // 3. Create appointment based on type
        switch (session.appointmentType) {
          case 'CONSULTATION':
            await createConsultationAppointment(tx, session, slot);
            break;
          case 'SUBSCRIPTION':
            await createSubscriptionAppointment(tx, session, slot);
            break;
          // ... handle other types
        }

        // 4. Mark session as completed
        await tx.checkoutSession.update({
          where: { id: session.id },
          data: { status: 'COMPLETED' }
        });
      });
    } catch (error) {
      // Handle specific errors
      if (error.message === 'SLOT_TAKEN') {
        await handleSlotTakenError(checkoutSessionId);
      } else {
        // Log error and mark for retry if possible
        await handleWebhookError(checkoutSessionId, error);
      }
    }
  }
}

// Error handling functions
async function handleSlotTakenError(sessionId: string) {
  await prisma.$transaction(async (tx) => {
    const session = await tx.checkoutSession.findUnique({
      where: { id: sessionId }
    });

    if (session.retryCount < 3) {
      // Try to find alternative slot
      const alternativeSlot = await findAlternativeSlot(tx, session);
      
      if (alternativeSlot) {
        await tx.checkoutSession.update({
          where: { id: sessionId },
          data: {
            slotId: alternativeSlot.id,
            retryCount: session.retryCount + 1
          }
        });
        // Retry webhook processing
        await processWebhook(sessionId);
      } else {
        // No alternative found - refund
        await initiateRefund(session.paymentIntentId);
      }
    } else {
      // Max retries reached - refund
      await initiateRefund(session.paymentIntentId);
    }
  });
}

// Cleanup job
export async function cleanupExpiredSessions() {
  const expiredSessions = await prisma.checkoutSession.findMany({
    where: {
      status: 'PENDING',
      expiresAt: { lt: new Date() }
    }
  });

  for (const session of expiredSessions) {
    await prisma.$transaction(async (tx) => {
      // Release slot
      await tx.slotOfAppointment.update({
        where: { id: session.slotId },
        data: { isTentative: false }
      });

      // Cancel payment intent if exists
      if (session.paymentIntentId) {
        await cancelPaymentIntent(session.paymentIntentId);
      }

      // Mark session as expired
      await tx.checkoutSession.update({
        where: { id: session.id },
        data: { status: 'EXPIRED' }
      });
    });
  }
}

  const [error, setError] = useState<string | null>(null);

  const handleCheckout = async () => {
    try {
      setStatus('processing');
      
      // 1. Initialize checkout
      const response = await fetch(`/api/checkout/register/${appointmentType}`, {
        method: 'POST',
        body: JSON.stringify({
          planId,
          slotIds,
          paymentGateway
        })
      });

      if (!response.ok) {
        if (response.status === 409) {
          setError('This slot is no longer available. Please choose another time.');
          return;
        }
        throw new Error('Checkout initialization failed');
      }

      const { clientSecret, sessionId } = await response.json();

      // 2. Handle payment
      const result = await stripe.confirmPayment({
        clientSecret,
        confirmParams: {
          return_url: `/success?session=${sessionId}`,
        }
      });

      if (result.error) {
        throw new Error(result.error.message);
      }

      setStatus('success');
    } catch (error) {
      setStatus('error');
      setError(error.message);
    }
  };

  // Cleanup on unmount
  useEffect(() => {
    return () => {
      if (status === 'processing') {
        // Call cleanup API
        fetch(`/api/checkout/cleanup/${sessionId}`);
      }
    };
  }, [status, sessionId]);

  return (
    <div>
      {status === 'processing' && (
        <LoadingOverlay message="Processing your booking..." />
      )}
      
      {error && (
        <ErrorAlert 
          message={error}
          action={handleCheckout}
          actionLabel="Try Again"
        />
      )}
      
      {/* Rest of the checkout UI */}
    </div>
  );
